CFG for ClusterOrientedProgrammingLanguage

PROGRAM -> ANNOTATION CLASS PROGRAM | E
ANNOTATION -> @Cluster( STRING_LITERAL )

CLASS -> class IDENTIFIER { CLASS_BODY }
CLASS_BODY -> METHOD CLASS_BODY
            | INITIALIZATION CLASS_BODY
            | DECLARATION CLASS_BODY
            | EXPRESSION CLASS_BODY
            | LOOPING_STATEMENT CLASS_BODY
            | CONDITIONAL_STATEMENT CLASS_BODY
            | E

METHOD -> method IDENTIFIER { METHOD_BODY }
METHOD_BODY -> INITIALIZATION METHOD_BODY
             | DECLARATION METHOD_BODY
             | EXPRESSION METHOD_BODY
             | LOOPING_STATEMENT METHOD_BODY
             | CONDITIONAL_STATEMENT METHOD_BODY
             | E

DECLARATION -> TYPE IDENTIFIER INITIALIZATION;
INITIALIZATION -> = LITERAL | E
LOOPING_STATEMENT -> for(LOOP_INITIALIZATION; CONDITION; INCREMENT) {LOOP_BODY}
                   | while(CONDITION) {LOOP_BODY}

CONDITIONAL_STATEMENT -> if(CONDITION) {CDTN_BODY} CONDITIONAL_STATEMENT'
CONDITIONAL_STATEMENT' -> else CONDITIONAL_STATEMENT'' | E
CONDITIONAL_STATEMENT'' -> {CDTN_BODY}
                         | CONDITIONAL_STATEMENT
                         | E

CONDITION -> IDENTIFIER COMPARISON_OPERATOR CONDITION_RHS
           | true
           | false

CONDITION_RHS -> IDENTIFIER | LITERAL

CDTN_BODY -> INITIALIZATION CDTN_BODY
           | DECLARATION CDTN_BODY
           | EXPRESSION CDTN_BODY
           | LOOPING_STATEMENT CDTN_BODY
           | CONDITIONAL_STATEMENT CDTN_BODY
           | E

LOOP_BODY -> INITIALIZATION LOOP_BODY
           | DECLARATION LOOP_BODY
           | EXPRESSION LOOP_BODY
           | LOOPING_STATEMENT LOOP_BODY
           | CONDITIONAL_STATEMENT LOOP_BODY
           | E

INCREMENT -> IDENTIFIER INC_OPERATOR
INC_OPERATOR -> ++ | --
COMPARISON_OPERATOR -> == | <= | >= | != | < | >
EXPRESSION -> IDENTIFIER EXPRESSION_TAIL
EXPRESSION_TAIL -> ASSIGNMENT_OPERATOR EXPRESSION_RHS ;
                 | INC_OPERATOR ;
                 | ;
EXPRESSION_RHS -> IDENTIFIER | LITERAL

LOOP_INITIALIZATION -> TYPE IDENTIFIER = LITERAL
ASSIGNMENT_OPERATOR -> = | += | -= | /= | *= | %=
TYPE -> int | float | string
LITERAL -> INTEGER_LITERAL | FLOAT_LITERAL | STRING_LITERAL

IDENTIFIER -> [a-zA-Z][a-zA-Z0-9]*
FLOAT_LITERAL -> [0-9]+\.[0-9]+
INTEGER_LITERAL -> [0-9][0-9]*
STRING_LITERAL -> ".*"
