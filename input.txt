int temperature, humidity, pressure;
float adjustedTemperature, riskScore;
int sensorCount;

sensorCount = 10;
temperature = 32;
humidity = 75;
pressure = 1012;

temperature++;
humidity--;
++pressure;

adjustedTemperature = (temperature * 1.8 + 32) - 2;
riskScore = adjustedTemperature * humidity / sensorCount;

temperature += 5;
humidity -= 10;
pressure *= 2;
pressure /= 4;
pressure %= 100;

riskScore = (temperature + humidity) * pressure / sensorCount;
